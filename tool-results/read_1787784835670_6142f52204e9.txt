    49→    49→    webkitSpeechRecognition?: SpeechRecognitionCtor;

    54→    54→function rms(analyser: AnalyserNode, data: Uint8Array<ArrayBuffer>): number {
    55→    55→  analyser.getByteTimeDomainData(data);
    56→    56→  let sum = 0;
    57→    57→  for (let i = 0; i < data.length; i++) {
    58→    58→    const v = (data[i] - 128) / 128;
    59→    59→    sum += v * v;
    60→    60→  }
    61→    61→  return Math.min(1, Math.sqrt(sum / data.length) * 3.4);
    62→    62→}

   156→   156→    sess.micAnalyser = null;

   198→   198→      setError(reason);

   203→   203→        setLastReply("");

   233→   233→    beginListeningRef.current = beginListening;

   239→   239→    const s = useInfinity.getState().settings;
   240→   240→    const history = [...historyRef.current, { role: "user" as const, content: userText }];
   241→   241→    const ac = new AbortController();

   244→   244→    try {
   245→   245→      const res = await fetch("/api/chat", {
   246→   246→        method: "POST",
   247→   247→        headers: { "Content-Type": "application/json" },

   277→   277→  /* ----------------------------- speak ------------------------------ */
   278→   278→
   279→   279→  const speak = useCallback(
   280→   280→    async (text: string): Promise<void> => {
   281→   281→      setAgentState("speaking");
   282→   282→      const s = useInfinity.getState().settings;
   283→   283→      const ac = abortRef.current;
   284→   284→      let buf: ArrayBuffer;
   285→   285→      try {
   286→   286→        const res = await fetch("/api/tts", {
   287→   287→          method: "POST",
   288→   288→          headers: { "Content-Type": "application/json" },
   289→   289→          signal: ac?.signal,
   290→   290→          body: JSON.stringify({ text, voice: s.voice, rate: s.rate }),
   291→   291→        });
   292→   292→        if (!res.ok) {
   293→   293→          const j = (await res.json().catch(() => null)) as { error?: string } | null;
   294→   294→          throw new Error(j?.error || "Voice synthesis failed.");
   295→   295→        }
   296→   296→        buf = await res.arrayBuffer();
   297→   297→      } catch (err) {
   298→   298→        if (ac?.signal.aborted) throw new DOMException("Aborted", "AbortError");
   299→   299→        if (err instanceof TypeError) throw new Error("Could not reach the voice service.");
   300→   300→        throw err;
   301→   301→      }
   302→   302→
   303→   303→      let ctx = ctxRef.current;
   304→   304→      if (!ctx) {
   305→   305→        ctx = new AudioContext();
   306→   306→        ctxRef.current = ctx;
   307→   307→      }
   308→   308→      try {
   309→   309→        await ctx.resume();
   310→   310→      } catch {
   311→   311→        /* will fall back to <audio> if blocked */
   312→   312→      }
   313→   313→      const ctx2 = ctx;
   314→   314→
   315→   315→      const playBuffer = () =>
   316→   316→        new Promise<void>((resolve, reject) => {
   317→   317→          let settled = false;
   318→   318→          let watchdog = 0;
   319→   319→          const finish = () => {
   320→   320→            if (settled) return;
   321→   321→            settled = true;
   322→   322→            if (watchdog) window.clearTimeout(watchdog);
   323→   323→            sourceRef.current = null;
   324→   324→            playbackAnalyserRef.current = null;
   325→   325→            resolve();
   326→   326→          };
   327→   327→          void (async () => {
   328→   328→            try {
   329→   329→              const audio = await ctx2.decodeAudioData(buf);
   330→   330→              const src = ctx2.createBufferSource();
   331→   331→              src.buffer = audio;
   332→   332→              const an = ctx2.createAnalyser();
   333→   333→              an.fftSize = 512;
   334→   334→              src.connect(an);
   335→   335→              an.connect(ctx2.destination);
   336→   336→              sourceRef.current = src;
   337→   337→              playbackAnalyserRef.current = an;
   338→   338→              playbackDataRef.current = new Uint8Array(new ArrayBuffer(an.fftSize));
   339→   339→              src.onended = finish;
   340→   340→              watchdog = window.setTimeout(
   341→   341→                finish,
   342→   342→                Math.max(1500, (audio.duration || 2) * 1000 + 1200)
   343→   343→              );
   344→   344→              src.start();
   345→   345→            } catch (e) {
   346→   346→              finish();
   347→   347→              reject(e instanceof Error ? e : new Error("Audio decoding failed."));
   348→   348→            }
   349→   349→          })();
   350→   350→        });
   351→   351→
   352→   352→      const playFallback = () =>
   353→   353→        new Promise<void>((resolve) => {
   354→   354→          const url = URL.createObjectURL(new Blob([buf], { type: "audio/mpeg" }));
   355→   355→          const el = new Audio(url);
   356→   356→          fallbackRef.current = el;
   357→   357→          syntheticUntilRef.current = Number.MAX_SAFE_INTEGER;
   358→   358→          let settled = false;
   359→   359→          let watchdog = 0;
   360→   360→          const finish = () => {
   361→   361→            if (settled) return;
   362→   362→            settled = true;
   363→   363→            if (watchdog) window.clearTimeout(watchdog);
   364→   364→            URL.revokeObjectURL(url);
   365→   365→            fallbackRef.current = null;
   366→   366→            syntheticUntilRef.current = 0;
   367→   367→            resolve();
   368→   368→          };
   369→   369→          watchdog = window.setTimeout(finish, 60000);
   370→   370→          el.onended = finish;
   371→   371→          el.onerror = finish;
   372→   372→          void el.play().catch(() => {
   373→   373→            // Autoplay blocked — end the turn so the loop still continues.
   374→   374→            finish();
   375→   375→          });
   376→   376→        });
   377→   377→
   378→   378→      await playBuffer().catch(() => playFallback());
   379→   379→    },
   380→   380→    [setAgentState]
   381→   381→  );
   382→   382→
   383→   383→  /* -------------------------- workbench ----------------------------- */
   384→   384→
   385→   385→  /** Execute a workbench command: toggle the grid, confirm out loud,
   386→   386→   *  then resume the conversation loop (never hits the LLM). */
   387→   387→  const applyWorkbench = useCallback(
   388→   388→    async (action: WorkbenchAction) => {
   389→   389→      const opening = action === "open";
   390→   390→      useInfinity.getState().setWorkbench(opening);
   391→   391→
   392→   392→      // Stop recognition first so Infinity never hears its own confirmation.
   393→   393→      const sess = sessionRef.current;
   394→   394→      if (sess) {
   395→   395→        try {
   396→   396→          sess.rec?.stop();
   397→   397→        } catch {
   398→   398→          /* noop */
   399→   399→        }
   400→   400→        if (sess.debounce) clearTimeout(sess.debounce);
   401→   401→        if (sess.interimTimer) clearTimeout(sess.interimTimer);
   402→   402→        sess.finalBuf = "";
   403→   403→        sess.lastInterim = "";
   404→   404→      }
   405→   405→
   406→   406→      if (activeRef.current) {
   407→   407→        const confirmText = opening ? "Workbench online." : "Workbench closed.";
   408→   408→        setLastReply(confirmText);
   409→   409→        try {
   410→   410→          await speak(confirmText);
   411→   411→        } catch {
   412→   412→          /* confirmation is best-effort */
   413→   413→        }
   414→   414→        if (!activeRef.current) return;
   415→   415→        const s2 = sessionRef.current;
   416→   416→        if (s2 && !s2.stopped) {
   417→   417→          window.setTimeout(() => {
   418→   418→            const s3 = sessionRef.current;
   419→   419→            if (s3 === s2 && !s3.stopped && activeRef.current) {
   420→   420→              setAgentState("listening");
   421→   421→              beginListeningRef.current(s3);
   422→   422→            }
   423→   423→          }, 300);
   424→   424→        } else {
   425→   425→          setAgentState("idle");
   426→   426→        }
   427→   427→      }
   428→   428→    },
   429→   429→    [setAgentState, speak]
   430→   430→  );
   431→   431→
   432→   432→  /** Returns true when the text was a workbench command (already handled). */
   433→   433→  const tryWorkbench = useCallback(
   434→   434→    (raw: string): boolean => {
   435→   435→      const action = matchWorkbenchCommand(raw);
   436→   436→      if (!action) return false;
   437→   437→      setInterim("");
   438→   438→      setLastUser(raw.trim());
   439→   439→      setError(null);
   440→   440→      void applyWorkbench(action);
   441→   441→      return true;
   442→   442→    },
   443→   443→    [applyWorkbench]
   444→   444→  );
   445→   445→
   446→   446→  /** Pause mic + clear buffers so Infinity never hears its own voice. */
   447→   447→  const pauseMic = useCallback(() => {
   448→   448→    const sess = sessionRef.current;
   449→   449→    if (!sess) return;
   450→   450→    try {
   451→   451→      sess.rec?.stop();
   452→   452→    } catch {
   453→   453→      /* noop */
   454→   454→    }
   455→   455→    if (sess.debounce) clearTimeout(sess.debounce);
   456→   456→    if (sess.interimTimer) clearTimeout(sess.interimTimer);
   457→   457→    sess.finalBuf = "";
   458→   458→    sess.lastInterim = "";
   459→   459→    setInterim("");
   460→   460→  }, []);
   461→   461→
   462→   462→  /** Back to listening (voice) or idle (text) after a workbench action. */
   463→   463→  const resumeAfterAction = useCallback(() => {
   464→   464→    if (!activeRef.current) return;
   465→   465→    const sess = sessionRef.current;
   466→   466→    if (sess && !sess.stopped) {
   467→   467→      window.setTimeout(() => {
   468→   468→        const s3 = sessionRef.current;
   469→   469→        if (s3 === sess && !s3.stopped && activeRef.current) {
   470→   470→          setAgentState("listening");
   471→   471→          beginListeningRef.current(s3);
   472→   472→        }
   473→   473→      }, 300);
   474→   474→    } else {
   475→   475→      setAgentState("idle");
   476→   476→    }
   477→   477→  }, [setAgentState]);
   478→   478→
   479→   479→  /* --------------------- build & delete models --------------------- */
   480→   480→
   481→   481→  /** Ask the LLM for a JSON model spec (single-shot, generous tokens). */
   482→   482→  const askSpec = useCallback(async (object: string): Promise<HoloSpec> => {
   483→   483→    const s = useInfinity.getState().settings;
   484→   484→    const ac = new AbortController();
   485→   485→    abortRef.current?.abort();
   486→   486→    abortRef.current = ac;
   487→   487→    const res = await fetch("/api/chat", {
   488→   488→      method: "POST",
   489→   489→      headers: { "Content-Type": "application/json" },
   490→   490→      signal: ac.signal,
   491→   491→      body: JSON.stringify({
   492→   492→        provider: s.provider,
   493→   493→        apiKey: s.apiKey.trim(),
   494→   494→        baseUrl: s.baseUrl.trim() || undefined,
   495→   495→        model: s.model.trim(),
   496→   496→        messages: [{ role: "user", content: `Build a holographic model of: ${object}` }],
   497→   497→        systemPrompt: MODEL_GEN_SYSTEM,
   498→   498→        maxTokens: 3000,
   499→   499→      }),
   500→   500→    });
   501→   501→    const data = (await res.json()) as
   502→   502→      | { ok: true; reply: string }
   503→   503→      | { ok: false; error: string };
   504→   504→    if (!res.ok || !data.ok) {
   505→   505→      throw new Error(data.ok ? "" : data.error || "The model generator failed.");
   506→   506→    }
   507→   507→    return parseHoloSpec(data.reply, object);
   508→   508→  }, []);
   509→   509→
   510→   510→  const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
   511→   511→
   512→   512→  /** Intercept "build a model of X" — speaks, shows progress, spawns model. */
   513→   513→  const tryBuild = useCallback(
   514→   514→    (raw: string): boolean => {
   515→   515→      const cmd = matchBuildCommand(raw);
   516→   516→      if (!cmd) return false;
   517→   517→
   518→   518→      setInterim("");
   519→   519→      setLastUser(raw.trim());
   520→   520→      setError(null);
   521→   521→
   522→   522→      const s = useInfinity.getState().settings;
   523→   523→      if (!isConfigured(s)) {
   524→   524→        toast("Add your API key in Settings to build models.", {
   525→   525→          action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   526→   526→        });
   527→   527→        onNeedSettingsRef.current();
   528→   528→        return true;
   529→   529→      }
   530→   530→
   531→   531→      const models = useInfinity.getState().models;
   532→   532→      if (models.length >= MAX_MODELS) {
   533→   533→        toast.error("The workbench is full — delete a model first.");
   534→   534→        return true;
   535→   535→      }
   536→   536→
   537→   537→      useInfinity.getState().setWorkbench(true);
   538→   538→      pauseMic();
   539→   539→
   540→   540→      void (async () => {
   541→   541→        try {
   542→   542→          await speak("OK, building that now.");
   543→   543→        } catch {
   544→   544→          /* best effort */
   545→   545→        }
   546→   546→        if (!activeRef.current && !sessionRef.current) {
   547→   547→          setAgentState("idle");
   548→   548→        }
   549→   549→        setBuilding({ name: cmd.object, phase: "building" });
   550→   550→
   551→   551→        try {
   552→   552→          const spec = await askSpec(cmd.object);
   553→   553→          const store = useInfinity.getState();
   554→   554→          if (store.models.length >= MAX_MODELS) throw new Error("Workbench is full.");
   555→   555→          store.addModel({
   556→   556→            id: `m-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   557→   557→            name: spec.name,
   558→   558→            spec,
   559→   559→            pos: nextSlot(store.models.length),
   560→   560→            rot: { x: 0.12, y: -0.55 },
   561→   561→          });
   562→   562→          setBuilding((b) => (b ? { ...b, name: spec.name, phase: "done" } : b));
   563→   563→          await sleep(900);
   564→   564→          setBuilding(null);
   565→   565→          try {
   566→   566→            await speak(`${spec.name} ready.`);
   567→   567→          } catch {
   568→   568→            /* best effort */
   569→   569→          }
   570→   570→        } catch (err) {
   571→   571→          if (abortRef.current?.signal.aborted) {
   572→   572→            setBuilding(null);
   573→   573→            return;
   574→   574→          }
   575→   575→          const msg = err instanceof Error && err.message ? err.message : "The build failed.";
   576→   576→          setError(msg);
   577→   577→          toast.error(msg);
   578→   578→          setBuilding((b) => (b ? { ...b, phase: "error" } : b));
   579→   579→          await sleep(1400);
   580→   580→          setBuilding(null);
   581→   581→          try {
   582→   582→            await speak("I couldn't build that.");
   583→   583→          } catch {
   584→   584→            /* best effort */
   585→   585→          }
   586→   586→        }
   587→   587→        resumeAfterAction();
   588→   588→      })();
   589→   589→      return true;
   590→   590→    },
   591→   591→    [askSpec, pauseMic, resumeAfterAction, setAgentState, speak]
   592→   592→  );
   593→   593→
   594→   594→  /** Intercept "delete the X" / "clear the workbench" while in workbench. */
   595→   595→  const tryDelete = useCallback(
   596→   596→    (raw: string): boolean => {
   597→   597→      const store = useInfinity.getState();
   598→   598→      if (!store.workbench || store.models.length === 0) return false;
   599→   599→      const cmd = matchDeleteCommand(
   600→   600→        raw,
   601→   601→        store.models.map((m) => m.name)
   602→   602→      );
   603→   603→      if (!cmd) return false;
   604→   604→
   605→   605→      setInterim("");
   606→   606→      setLastUser(raw.trim());
   607→   607→      setError(null);
   608→   608→      pauseMic();
   609→   609→
   610→   610→      if (cmd.all) {
   611→   611→        const names = store.models.map((m) => m.name);
   612→   612→        store.clearModels();
   613→   613→        void (async () => {
   614→   614→          try {
   615→   615→            await speak(
   616→   616→              names.length === 1
   617→   617→                ? `Removed the ${names[0]}.`
   618→   618→                : `Cleared ${names.length} models from the workbench.`
   619→   619→            );
   620→   620→          } catch {
   621→   621→            /* best effort */
   622→   622→          }
   623→   623→          resumeAfterAction();
   624→   624→        })();
   625→   625→        return true;
   626→   626→      }
   627→   627→
   628→   628→      const target = store.models.find(
   629→   629→        (m) => m.name.toLowerCase() === cmd.name.toLowerCase()
   630→   630→      ) ?? store.models.find((m) => m.name.toLowerCase().includes(cmd.name.toLowerCase()));
   631→   631→      if (!target) return false; // mentioned nothing we have → normal chat
   632→   632→      store.removeModel(target.id);
   633→   633→      void (async () => {
   634→   634→        try {
   635→   635→          await speak(`Removed the ${target.name}.`);
   636→   636→        } catch {
   637→   637→          /* best effort */
   638→   638→        }
   639→   639→        resumeAfterAction();
   640→   640→      })();
   641→   641→      return true;
   642→   642→    },
   643→   643→    [pauseMic, resumeAfterAction, speak]
   644→   644→  );
   645→   645→
   646→   646→  /* ---------------------------- one turn ---------------------------- */
   647→   647→
   648→   648→  const runTurn = useCallback(
   649→   649→    async (userText: string) => {
   650→   650→      if (stateRef.current === "thinking" || stateRef.current === "speaking") return;
   651→   651→      if (tryWorkbench(userText)) return;
   652→   652→      if (tryDelete(userText)) return;
   653→   653→      if (tryBuild(userText)) return;
   654→   654→      setError(null);
   655→   655→      setLastUser(userText);
   656→   656→      setInterim("");
   657→   657→      setAgentState("thinking");
   658→   658→
   659→   659→      const sessNow = sessionRef.current;
   660→   660→      if (sessNow?.rec) {
   661→   661→        try {
   662→   662→          sessNow.rec.stop();
   663→   663→        } catch {
   664→   664→          /* noop */
   665→   665→        }
   666→   666→      }
   667→   667→
   668→   668→      let reply = "";
   669→   669→      try {
   670→   670→        reply = await askChat(userText);
   671→   671→      } catch (err) {
   672→   672→        if (!activeRef.current || abortRef.current?.signal.aborted) return;
   673→   673→        const msg =
   674→   674→          err instanceof Error && err.message ? err.message : "The AI service failed.";
   675→   675→        setError(msg);
   676→   676→        toast.error(msg, {
   677→   677→          action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   678→   678→        });
   679→   679→        const s2 = sessionRef.current;
   680→   680→        if (s2 && !s2.stopped) {
   681→   681→          setAgentState("listening");
   682→   682→          beginListeningRef.current(s2);
   683→   683→        } else {
   684→   684→          setAgentState("idle");
   685→   685→        }
   686→   686→        return;
   687→   687→      }
   688→   688→
   689→   689→      if (!activeRef.current) return;
   690→   690→      setLastReply(reply);
   691→   691→
   692→   692→      try {
   693→   693→        await speak(reply);
   694→   694→      } catch (err) {
   695→   695→        if (!activeRef.current || abortRef.current?.signal.aborted) return;
   696→   696→        const msg = err instanceof Error && err.message ? err.message : "Voice playback failed.";
   697→   697→        setError(msg);
   698→   698→        toast.error(msg);
   699→   699→      }
   700→   700→      if (!activeRef.current) return;
   701→   701→
   702→   702→      // Resume: voice sessions go back to listening; text sessions wait for input.
   703→   703→      const sess = sessionRef.current;
   704→   704→      if (sess && !sess.stopped) {
   705→   705→        const delay = window.setTimeout(() => {
   706→   706→          const s3 = sessionRef.current;
   707→   707→          if (s3 === sess && !s3.stopped && activeRef.current) {
   708→   708→            setAgentState("listening");
   709→   709→            beginListeningRef.current(s3);
   710→   710→          }
   711→   711→        }, 350);
   712→   712→        void delay;
   713→   713→      } else {
   714→   714→        setAgentState("idle");
   715→   715→      }
   716→   716→    },
   717→   717→    [askChat, setAgentState, speak, tryBuild, tryDelete, tryWorkbench]
   718→   718→  );
   719→   719→
   720→   720→  /* --------------------- recognition wiring ------------------------- */
   721→   721→
   722→   722→  const setupRecognition = useCallback(
   723→   723→    (mySess: VoiceSession) => {
   724→   724→      const rec = new mySess.recCtor();
   725→   725→      rec.continuous = true;
   726→   726→      rec.interimResults = true;
   727→   727→      rec.lang = "en-US";
   728→   728→
   729→   729→      const submitVoice = (text: string) => {
   730→   730→        const t = text.trim();
   731→   731→        if (!t) return;
   732→   732→        if (mySess.interimTimer) {
   733→   733→          clearTimeout(mySess.interimTimer);
   734→   734→          mySess.interimTimer = null;
   735→   735→        }
   736→   736→        if (mySess.debounce) {
   737→   737→          clearTimeout(mySess.debounce);
   738→   738→          mySess.debounce = null;
   739→   739→        }
   740→   740→        mySess.finalBuf = "";
   741→   741→        mySess.lastInterim = "";
   742→   742→        setInterim("");
   743→   743→        try {
   744→   744→          mySess.rec?.stop();
   745→   745→        } catch {
   746→   746→          /* noop */
   747→   747→        }
   748→   748→        void runTurn(t);
   749→   749→      };
   750→   750→
   751→   751→      rec.onresult = (event) => {
   752→   752→        if (sessionRef.current !== mySess || mySess.stopped) return;
   753→   753→        if (stateRef.current !== "listening") return;
   754→   754→        let interimText = "";
   755→   755→        for (let i = event.resultIndex; i < event.results.length; i++) {
   756→   756→          const r = event.results[i];
   757→   757→          if (r.isFinal) mySess.finalBuf += r[0].transcript + " ";
   758→   758→          else interimText += r[0].transcript;
   759→   759→        }
   760→   760→        if (interimText) setInterim(interimText);
   761→   761→        if (mySess.interimTimer) {
   762→   762→          clearTimeout(mySess.interimTimer);
   763→   763→          mySess.interimTimer = null;
   764→   764→        }
   765→   765→        mySess.lastInterim = interimText;
   766→   766→
   767→   767→        if (mySess.finalBuf.trim()) {
   768→   768→          if (mySess.debounce) clearTimeout(mySess.debounce);
   769→   769→          // Small pause after final words → send.
   770→   770→          mySess.debounce = setTimeout(() => submitVoice(mySess.finalBuf), 850);
   771→   771→        } else if (interimText.trim()) {
   772→   772→          // Safety net: some browsers stall before emitting a final result.
   773→   773→          mySess.interimTimer = setTimeout(
   774→   774→            () => submitVoice(mySess.lastInterim),
   775→   775→            1700
   776→   776→          );
   777→   777→        }
   778→   778→      };
   779→   779→
   780→   780→      rec.onerror = (event) => {
   781→   781→        if (sessionRef.current !== mySess || mySess.stopped) return;
   782→   782→        const kind = event.error;
   783→   783→        if (kind === "no-speech" || kind === "aborted") return;
   784→   784→        if (kind === "not-allowed" || kind === "service-not-allowed") {
   785→   785→          fallbackToText(
   786→   786→            "Microphone access was blocked — open this app in its own browser tab to talk out loud, or type below."
   787→   787→          );
   788→   788→          return;
   789→   789→        }
   790→   790→        if (kind === "network") {
   791→   791→          fallbackToText(
   792→   792→            "Speech recognition lost its connection — keep talking by typing below."
   793→   793→          );
   794→   794→        }
   795→   795→      };
   796→   796→
   797→   797→      rec.onend = () => {
   798→   798→        if (sessionRef.current !== mySess || mySess.stopped) return;
   799→   799→        if (stateRef.current === "listening" && modeRef.current === "voice") {
   800→   800→          beginListeningRef.current(mySess);
   801→   801→        }
   802→   802→      };
   803→   803→
   804→   804→      mySess.rec = rec;
   805→   805→    },
   806→   806→    [fallbackToText, runTurn]
   807→   807→  );
   808→   808→
   809→   809→  /* ------------------------------ start ----------------------------- */
   810→   810→
   811→   811→  const start = useCallback(() => {
   812→   812→    if (activeRef.current) return;
   813→   813→    const s = useInfinity.getState().settings;
   814→   814→    if (!isConfigured(s)) {
   815→   815→      toast("Add your API key in Settings to start talking to Infinity.", {
   816→   816→        action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   817→   817→      });
   818→   818→      onNeedSettingsRef.current();
   819→   819→      return;
   820→   820→    }
   821→   821→    const recCtor = getSpeechRecognitionCtor();
   822→   822→    if (!recCtor) {
   823→   823→      toast("Voice input isn't supported in this browser — type instead (press /).");
   824→   824→      fallbackToText(
   825→   825→        "Voice input isn't supported in this browser — type below and Infinity will answer out loud."
   826→   826→      );
   827→   827→      return;
   828→   828→    }
   829→   829→
   830→   830→    const sess: VoiceSession = {
   831→   831→      stopped: false,
   832→   832→      micStream: null,
   833→   833→      micAnalyser: null,
   834→   834→      micData: null,
   835→   835→      rec: null,
   836→   836→      recCtor,
   837→   837→      finalBuf: "",
   838→   838→      lastInterim: "",
   839→   839→      interimTimer: null,
   840→   840→      debounce: null,
   841→   841→      lastRestart: 0,
   842→   842→    };
   843→   843→    sessionRef.current = sess;
   844→   844→    activeRef.current = true;
   845→   845→    modeRef.current = "voice";
   846→   846→    setMode("voice");
   847→   847→    setSessionActive(true);
   848→   848→    setMicBlocked(false);
   849→   849→    setError(null);
   850→   850→    setLastUser("");
   851→   851→    setLastReply("");
   852→   852→    historyRef.current = [];
   853→   853→
   854→   854→    void (async () => {
   855→   855→      try {
   856→   856→        let ctx = ctxRef.current;
   857→   857→        if (!ctx) {
   858→   858→          ctx = new AudioContext();
   859→   859→          ctxRef.current = ctx;
   860→   860→        }
   861→   861→        try {
   862→   862→          await ctx.resume();
   863→   863→        } catch {
   864→   864→          /* noop */
   865→   865→        }
   866→   866→        sess.micStream = await navigator.mediaDevices.getUserMedia({
   867→   867→          audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
   868→   868→        });
   869→   869→        if (sessionRef.current !== sess) {
   870→   870→          sess.micStream.getTracks().forEach((t) => t.stop());
   871→   871→          return;
   872→   872→        }
   873→   873→        const micSrc = ctx.createMediaStreamSource(sess.micStream);
   874→   874→        const an = ctx.createAnalyser();
   875→   875→        an.fftSize = 512;
   876→   876→        micSrc.connect(an); // deliberately not connected to destination
   877→   877→        sess.micAnalyser = an;
   878→   878→        sess.micData = new Uint8Array(new ArrayBuffer(an.fftSize));
   879→   879→
   880→   880→        setupRecognition(sess);
   881→   881→        setAgentState("listening");
   882→   882→        beginListening(sess);
   883→   883→      } catch {
   884→   884→        if (sessionRef.current !== sess) return;
   885→   885→        fallbackToText(
   886→   886→          "Microphone unavailable — open this app in its own browser tab and allow the mic, or type below."
   887→   887→        );
   888→   888→        toast.error("Microphone unavailable — you can type instead.");
   889→   889→      }
   890→   890→    })();
   891→   891→  }, [beginListening, fallbackToText, setAgentState, setupRecognition]);
   892→   892→
   893→   893→  /* ----------------------------- typing ----------------------------- */
   894→   894→
   895→   895→  const sendText = useCallback(
   896→   896→    (text: string) => {
   897→   897→      const t = text.trim();
   898→   898→      if (!t) return;
   899→   899→      // Workbench commands are local — they work even without an API key.
   900→   900→      if (tryWorkbench(t)) return;
   901→   901→      if (tryDelete(t)) return;
   902→   902→      if (tryBuild(t)) return;
   903→   903→      const s = useInfinity.getState().settings;
   904→   904→      if (!isConfigured(s)) {
   905→   905→        toast("Add your API key in Settings first.", {
   906→   906→          action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   907→   907→        });
   908→   908→        onNeedSettingsRef.current();
   909→   909→        return;
   910→   910→      }
   911→   911→      if (!activeRef.current) {
   912→   912→        activeRef.current = true;
   913→   913→        modeRef.current = "text";
   914→   914→        setMode("text");
   915→   915→        setSessionActive(true);
   916→   916→        setMicBlocked(false);
   917→   917→        setError(null);
   918→   918→        setLastUser("");
   919→   919→        setLastReply("");
   920→   920→        historyRef.current = [];
   921→   921→        setAgentState("idle");
   922→   922→      }
   923→   923→      void runTurn(t);
   924→   924→    },
   925→   925→    [runTurn, setAgentState, tryBuild, tryDelete, tryWorkbench]
   926→   926→  );
   927→   927→
   928→   928→  const toggle = useCallback(() => {
   929→   929→    if (activeRef.current) stop();
   930→   930→    else start();
   931→   931→  }, [start, stop]);
   932→   932→
   933→   933→  /* --------------------------- level loop --------------------------- */
   934→   934→
   935→   935→  useEffect(() => {
   936→   936→    let raf = 0;
   937→   937→    const frame = (t: number) => {
   938→   938→      const st = stateRef.current;
   939→   939→      let target = 0;
   940→   940→      const sess = sessionRef.current;
   941→   941→      if (st === "listening" && sess?.micAnalyser && sess.micData) {
   942→   942→        target = rms(sess.micAnalyser, sess.micData);
   943→   943→      } else if (st === "speaking") {
   944→   944→        if (playbackAnalyserRef.current && playbackDataRef.current) {
   945→   945→          target = rms(playbackAnalyserRef.current, playbackDataRef.current);
   946→   946→        } else if (syntheticUntilRef.current > Date.now()) {
   947→   947→          target = 0.32 + 0.28 * Math.sin(t * 0.012);
   948→   948→        }
   949→   949→      }
   950→   950→      const prev = levelRef.current;
   951→   951→      levelRef.current = prev + (target - prev) * (target > prev ? 0.55 : 0.12);
   952→   952→      raf = requestAnimationFrame(frame);
   953→   953→    };
   954→   954→    raf = requestAnimationFrame(frame);
   955→   955→    return () => cancelAnimationFrame(raf);
   956→   956→  }, []);
   957→   957→
   958→   958→  // stop everything on unmount
   959→   959→  useEffect(() => () => stopRef.current(), []);
   960→   960→
   961→   961→  return {
   962→   962→    state,
   963→   963→    mode,
   964→   964→    sessionActive,
   965→   965→    interim,
   966→   966→    lastUser,
   967→   967→    lastReply,
   968→   968→    error,
   969→   969→    micBlocked,
   970→   970→    building,
   971→   971→    levelRef,
   972→   972→    start,
   973→   973→    stop,
   974→   974→    toggle,
   975→   975→    sendText,
   976→   976→  };
   977→   977→}
