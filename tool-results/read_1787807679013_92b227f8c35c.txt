    20→/**
    21→ * Never blind-`res.json()`: a gateway/proxy error page is HTML, and
    22→ * WebKit's raw rejection for that is the cryptic
    23→ * "The string did not match the pattern expected." Read text, parse safely.
    24→ */
    25→async function safeJson<T>(res: Response): Promise<T | null> {
    26→  try {
    27→    const text = await res.text();
    28→    if (!text) return null;
    29→    return JSON.parse(text) as T;
    30→  } catch {
    31→    return null;
    32→  }
    33→}

    64→function getSpeechRecognitionCtor(): SpeechRecognitionCtor | null {
    65→  if (typeof window === "undefined") return null;
    66→  const w = window as unknown as {
    67→    SpeechRecognition?: SpeechRecognitionCtor;
    68→    webkitSpeechRecognition?: SpeechRecognitionCtor;
    69→  };
    70→  return w.SpeechRecognition ?? w.webkitSpeechRecognition ?? null;
    71→}

    73→function rms(analyser: AnalyserNode, data: Uint8Array<ArrayBuffer>): number {
    74→  analyser.getByteTimeDomainData(data);
    75→  let sum = 0;
    76→  for (let i = 0; i < data.length; i++) {
    77→    const v = (data[i] - 128) / 128;
    78→    sum += v * v;
    79→  }
    80→  return Math.min(1, Math.sqrt(sum / data.length) * 3.4);
    81→}

   143→  const stateRef = useRef<AgentState>("idle");

   168→   */
   169→  const setAgentState = useCallback((s: AgentState) => {
   170→    stateRef.current = s;
   171→    setState(s);
   172→  }, []);
   173→
   174→  /* ---------------------------- teardown ---------------------------- */
   175→
   176→  const teardownVoice = useCallback(() => {
   177→    const sess = sessionRef.current;
   178→    if (!sess) return;
   179→    sess.stopped = true;
   180→    if (sess.debounce) clearTimeout(sess.debounce);
   181→    if (sess.interimTimer) clearTimeout(sess.interimTimer);
   182→    try {
   183→      sess.rec?.abort();
   184→    } catch {
   185→      /* already stopped */
   186→    }
   187→    sess.micStream?.getTracks().forEach((t) => t.stop());
   188→    sess.micAnalyser = null;
   189→    sess.micData = null;
   190→    sess.rec = null;
   191→    sessionRef.current = null;
   192→  }, []);
   193→
   194→  const stop = useCallback(() => {
   195→    abortRef.current?.abort();
   196→    teardownVoice();
   197→    try {
   198→      sourceRef.current?.stop();
   199→    } catch {
   200→      /* already stopped */

   211→    activeRef.current = false;
   212→    modeRef.current = null;

   214→    setMode(null);
   215→    setAgentState("idle");
   216→    setInterim("");

   221→  const stopRef = useRef(stop);
   222→  useEffect(() => {
   223→    stopRef.current = stop;
   224→  }, [stop]);
   225→
   226→  /** Kill the voice half but keep the conversation alive in text mode. */
   227→  const fallbackToText = useCallback(
   228→    (reason: string) => {
   229→      teardownVoice();

   231→      setError(reason);

   260→      sess.rec?.start();

   266→    beginListeningRef.current = beginListening;

   271→  const askChat = useCallback(async (userText: string): Promise<string> => {
   272→    const s = useInfinity.getState().settings;
   273→    const history = [...historyRef.current, { role: "user" as const, content: userText }];
   274→    const ac = new AbortController();

   310→  /* ----------------------------- speak ------------------------------ */
   311→
   312→  const speak = useCallback(
   313→    async (text: string): Promise<void> => {
   314→      setAgentState("speaking");

   316→      const ac = abortRef.current;

   319→        const res = await fetch("/api/tts", {

   330→      } catch (err) {
   331→        if (ac?.signal.aborted) throw new DOMException("Aborted", "AbortError");
   332→        if (err instanceof TypeError) throw new Error("Could not reach the voice service.");

   334→      }

   336→      let ctx = ctxRef.current;
   337→      if (!ctx) {
   338→        ctx = new AudioContext();
   339→        ctxRef.current = ctx;
   340→      }
   341→      try {
   342→        await ctx.resume();
   343→      } catch {
   344→        /* will fall back to <audio> if blocked */
   345→      }
   346→      const ctx2 = ctx;
   347→
   348→      const playBuffer = () =>
   349→        new Promise<void>((resolve, reject) => {
   350→          let settled = false;
   351→          let watchdog = 0;
   352→          const finish = () => {
   353→            if (settled) return;
   354→            settled = true;
   355→            if (watchdog) window.clearTimeout(watchdog);
   356→            sourceRef.current = null;
   357→            playbackAnalyserRef.current = null;
   358→            resolve();
   359→          };
   360→          void (async () => {
   361→            try {
   362→              const audio = await ctx2.decodeAudioData(buf);
   363→              const src = ctx2.createBufferSource();
   364→              src.buffer = audio;
   365→              const an = ctx2.createAnalyser();

   367→              src.connect(an);

   372→              src.onended = finish;
   373→              watchdog = window.setTimeout(
   374→                finish,
   375→                Math.max(1500, (audio.duration || 2) * 1000 + 1200)
   376→              );
   377→              src.start();
   378→            } catch (e) {
   379→              finish();
   380→              reject(e instanceof Error ? e : new Error("Audio decoding failed."));
   381→            }
   382→          })();
   383→        });
   384→
   385→      const playFallback = () =>
   386→        new Promise<void>((resolve) => {
   387→          const url = URL.createObjectURL(new Blob([buf], { type: "audio/mpeg" }));
   388→          const el = new Audio(url);
   389→          fallbackRef.current = el;
   390→          syntheticUntilRef.current = Number.MAX_SAFE_INTEGER;
   391→          let settled = false;
   392→          let watchdog = 0;
   393→          const finish = () => {
   394→            if (settled) return;
   395→            settled = true;
   396→            if (watchdog) window.clearTimeout(watchdog);
   397→            URL.revokeObjectURL(url);
   398→            fallbackRef.current = null;
   399→            syntheticUntilRef.current = 0;
   400→            resolve();
   401→          };
   402→          watchdog = window.setTimeout(finish, 60000);
   403→          el.onended = finish;
   404→          el.onerror = finish;
   405→          void el.play().catch(() => {
   406→            // Autoplay blocked — end the turn so the loop still continues.
   407→            finish();
   408→          });
   409→        });
   410→
   411→      await playBuffer().catch(() => playFallback());
   412→    },
   413→    [setAgentState]
   414→  );

   416→  /* ------------------- respond / transcript ------------------------ */
   417→
   418→  /** Append a line to the visible conversation log. */
   419→  const pushTranscript = useCallback((role: "user" | "infinity", text: string) => {
   420→    const id = ++transcriptIdRef.current;
   421→    setTranscript((prev) => {
   422→      const next = [...prev, { id, role, text }];

   427→  /** Infinity's side of the conversation: SPOKEN in voice mode, shown as
   428→   *  silent text in typing mode — a typed session never plays any sound. */

   430→    async (text: string): Promise<void> => {
   431→      if (modeRef.current === "voice") {
   432→        await speak(text);
   433→        return;
   434→      }
   435→      setLastReply(text);
   436→      pushTranscript("infinity", text);
   437→    },
   438→    [pushTranscript, speak]
   439→  );
   440→
   441→  /** The user's side: caption line + transcript entry (both modes log; only
   442→   *  typing mode displays the transcript). */

   444→    (text: string) => {
   445→      setLastUser(text);
   446→      pushTranscript("user", text);
   447→    },

   456→    async (action: WorkbenchAction) => {
   457→      const opening = action === "open";
   458→      useInfinity.getState().setWorkbench(opening);
   459→
   460→      // Stop recognition first so Infinity never hears its own confirmation.

   462→      if (sess) {
   463→        try {
   464→          sess.rec?.stop();
   465→        } catch {
   466→          /* noop */
   467→        }
   468→        if (sess.debounce) clearTimeout(sess.debounce);
   469→        if (sess.interimTimer) clearTimeout(sess.interimTimer);
   470→        sess.finalBuf = "";
   471→        sess.lastInterim = "";
   472→      }
   473→
   474→      if (activeRef.current) {
   475→        const confirmText = opening ? "Workbench online." : "Workbench closed.";
   476→        setLastReply(confirmText);
   477→        try {
   478→          await respond(confirmText);
   479→        } catch {
   480→          /* confirmation is best-effort */
   481→        }
   482→        if (!activeRef.current) return;
   483→        const s2 = sessionRef.current;
   484→        if (s2 && !s2.stopped) {
   485→          window.setTimeout(() => {
   486→            const s3 = sessionRef.current;
   487→            if (s3 === s2 && !s3.stopped && activeRef.current) {
   488→              setAgentState("listening");
   489→              beginListeningRef.current(s3);
   490→            }
   491→          }, 300);
   492→        } else {
   493→          setAgentState("idle");
   494→        }
   495→      }
   496→    },
   497→    [respond, setAgentState]
   498→  );

   500→  /** Returns true when the text was a workbench command (already handled). */
   501→  const tryWorkbench = useCallback(
   502→    (raw: string): boolean => {
   503→      const action = matchWorkbenchCommand(raw);
   504→      if (!action) return false;
   505→      setInterim("");
   506→      noteUser(raw.trim());
   507→      setError(null);
   508→      void applyWorkbench(action);
   509→      return true;
   510→    },
   511→    [applyWorkbench, noteUser]

   514→  /** Pause mic + clear buffers so Infinity never hears its own voice. */
   515→  const pauseMic = useCallback(() => {
   516→    const sess = sessionRef.current;
   517→    if (!sess) return;
   518→    try {
   519→      sess.rec?.stop();
   520→    } catch {
   521→      /* noop */
   522→    }
   523→    if (sess.debounce) clearTimeout(sess.debounce);
   524→    if (sess.interimTimer) clearTimeout(sess.interimTimer);
   525→    sess.finalBuf = "";
   526→    sess.lastInterim = "";
   527→    setInterim("");
   528→  }, []);
   529→
   530→  /** Back to listening (voice) or idle (text) after a workbench action. */
   531→  const resumeAfterAction = useCallback(() => {
   532→    if (!activeRef.current) return;
   533→    const sess = sessionRef.current;
   534→    if (sess && !sess.stopped) {
   535→      window.setTimeout(() => {
   536→        const s3 = sessionRef.current;
   537→        if (s3 === sess && !s3.stopped && activeRef.current) {
   538→          setAgentState("listening");
   539→          beginListeningRef.current(s3);
   540→        }
   541→      }, 300);
   542→    } else {
   543→      setAgentState("idle");
   544→    }
   545→  }, [setAgentState]);

   547→  /* --------------------- build & delete models --------------------- */
   548→
   549→  const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
   550→
   551→  /** Intercept "build a model of X" — LOCAL builders first (instant, offline,
   552→   *  keyless); the user's AI only designs objects the local system doesn't
   553→   *  know (or when explicitly asked to design/invent). A model always spawns.
   554→   *  Speaks, shows progress, persists. */
   555→  const tryBuild = useCallback(
   556→    (raw: string): boolean => {
   557→      const cmd = matchBuildCommand(raw, useInfinity.getState().workbench);
   558→      if (!cmd) return false;
   559→
   560→      setInterim("");
   561→      noteUser(raw.trim());
   562→      setError(null);
   563→
   564→      const models = useInfinity.getState().models;
   565→      if (models.length >= MAX_MODELS) {
   566→        toast.error("The workbench is full — delete a model first.");
   567→        return true;
   568→      }
   569→
   570→      useInfinity.getState().setWorkbench(true);
   571→      pauseMic();
   572→
   573→      void (async () => {
   574→        const settings = useInfinity.getState().settings;
   575→        const keyReady = isConfigured(settings);
   576→
   577→        // 1) LOCAL FIRST — phrases → library → families. Instant, offline,
   578→        //    deterministic. Skipped only when the user explicitly asks the
   579→        //    AI to design it AND a key is configured.
   580→        let spec: HoloSpec | null = null;
   581→        if (!cmd.forceDesign || !keyReady) {
   582→          spec =
   583→            matchPhraseModel(cmd.object) ??
   584→            matchLibraryModel(cmd.object) ??
   585→            matchFamilyModel(cmd.object);
   586→        }
   587→
   588→        // 2) Cached AI designs — an object the locals DON'T know, asked for
   589→        //    before: identical model, instantly, zero API calls.
   590→        if (!spec) spec = cacheGetSpec(cmd.object);
   591→
   592→        let designed = false;
   593→        let designFailed = false;
   594→        const willDesign = !spec && keyReady;
   595→
   596→        const spoken = respond(willDesign ? "OK, designing that now." : "OK, building that now.").catch(
   597→          () => {
   598→            /* best effort */
   599→          }
   600→        );
   601→
   602→        // 3) AI design — the user's LLM invents the object from scratch.
   603→        //    Only reached for objects the local builders don't recognize
   604→        //    (or when explicitly forced with "design a …").
   605→        if (willDesign) {
   606→          setBuilding({
   607→            name: cmd.object,
   608→            phase: "building",
   609→            progress: 0.06,
   610→            partsDone: 0,
   611→            count: null,
   612→            note: "DESIGNING",
   613→          });
   614→          // Gentle crawl while the model thinks (caps at 42%).
   615→          const crawl = window.setInterval(() => {
   616→            setBuilding((b) =>
   617→              b && b.note ? { ...b, progress: Math.min(0.42, b.progress + 0.005) } : b
   618→            );
   619→          }, 500);
   620→          try {
   621→            const out = await designHoloSpec({
   622→              object: cmd.object,
   623→              provider: settings.provider,
   624→              apiKey: settings.apiKey,
   625→              baseUrl: settings.baseUrl,
   626→              model: settings.model,
   627→              onProgress: (p) =>
   628→                setBuilding((b) =>
   629→                  b
   630→                    ? {
   631→                        ...b,
   632→                        note: p.phase === "thinking" ? "DESIGNING" : `${p.partsDesigned} PARTS`,
   633→                        progress: Math.max(b.progress, Math.min(0.45, p.progress)),
   634→                      }
   635→                    : b
   636→                ),
   637→            });
   638→            if (out?.spec) {
   639→              spec = out.spec;
   640→              designed = true;
   641→              if (!out.salvaged) cachePutSpec(cmd.object, out.spec);
   642→            }
   643→          } catch {
   644→            /* fall through to local builders */
   645→          } finally {
   646→            window.clearInterval(crawl);
   647→          }
   648→          if (!spec) designFailed = true;
   649→        }
   650→
   651→        // 4) Abstract archetypes guarantee a model — no key, no network,
   652→        //    nothing recognized: still spawns a seeded hologram.
   653→        if (!spec) {
   654→          spec = generateModel(cmd.object);
   655→        }
   656→
   657→        const finalSpec = spec;
   658→        const from = designed ? 0.45 : 0.08;
   659→
   660→        setBuilding({
   661→          name: finalSpec.name,
   662→          phase: "building",
   663→          progress: from,
   664→          partsDone: 0,
   665→          count: finalSpec.parts.length,
   666→        });
   667→
   668→        const store = useInfinity.getState();
   669→        store.addModel({
   670→          id: `m-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   671→          name: finalSpec.name,
   672→          spec: finalSpec,
   673→          pos: nextSlot(store.models.length),
   674→          rot: { x: 0.12, y: -0.55 },
   675→          bornAt: Date.now(),
   676→        });
   677→
   678→        // Progress tracks the actual on-screen part-by-part assembly.
   679→        const t0 = performance.now();
   680→        const iv = window.setInterval(() => {
   681→          // Assembly starts once the card's spawn animation has settled
   682→          // (the 3D canvas mounts then — see SPAWN_SETTLE_MS).
   683→          const frac = Math.min(
   684→            1,
   685→            Math.max(0, (performance.now() - t0 - SPAWN_SETTLE_MS) / ASSEMBLE_MS)
   686→          );
   687→          setBuilding((b) =>
   688→            b
   689→              ? {
   690→                  ...b,
   691→                  progress: from + (1 - from) * frac,
   692→                  partsDone: Math.round(frac * finalSpec.parts.length),
   693→                }
   694→              : b
   695→          );
   696→          if (frac >= 1) window.clearInterval(iv);
   697→        }, 90);
   698→
   699→        await spoken;
   700→        if (!activeRef.current && !sessionRef.current) {
   701→          setAgentState("idle");
   702→        }
   703→        await sleep(
   704→          Math.max(0, SPAWN_SETTLE_MS + ASSEMBLE_MS + 400 - (performance.now() - t0))
   705→        );
   706→        window.clearInterval(iv);
   707→
   708→        setBuilding((b) => (b ? { ...b, phase: "done", progress: 1 } : b));
   709→        await sleep(650);
   710→        setBuilding(null);
   711→        try {
   712→          await respond(`${finalSpec.name} ready.`);
   713→        } catch {
   714→          /* best effort */
   715→        }
   716→        if (designed) {
   717→          toast.success(`${finalSpec.name} — designed by AI`);
   718→        } else if (designFailed) {
   719→          toast("AI design unavailable — built a local model instead");
   720→        }
   721→        resumeAfterAction();
   722→      })().catch((err) => {
   723→        // Unreachable in practice — local geometry + speech can't fail.
   724→        // Keep it quiet and visible instead of ever saying "couldn't build".
   725→        console.error("build failed", err);
   726→        setBuilding(null);
   727→        toast.error("Something went wrong showing that model.");
   728→        resumeAfterAction();
   729→      });
   730→      return true;
   731→    },
   732→    [noteUser, pauseMic, respond, resumeAfterAction, setAgentState]
   733→  );
   734→
   735→  /** Intercept "delete the X" / "clear the workbench" while in workbench. */
   736→  const tryDelete = useCallback(
   737→    (raw: string): boolean => {
   738→      const store = useInfinity.getState();
   739→      if (!store.workbench || store.models.length === 0) return false;
   740→      const cmd = matchDeleteCommand(
   741→        raw,
   742→        store.models.map((m) => m.name)
   743→      );
   744→      if (!cmd) return false;
   745→
   746→      setInterim("");
   747→      noteUser(raw.trim());
   748→      setError(null);
   749→      pauseMic();
   750→
   751→      if ("all" in cmd) {
   752→        const names = store.models.map((m) => m.name);
   753→        store.clearModels();
   754→        void (async () => {
   755→          try {
   756→            await respond(
   757→              names.length === 1
   758→                ? `Removed the ${names[0]}.`
   759→                : `Cleared ${names.length} models from the workbench.`
   760→            );
   761→          } catch {
   762→            /* best effort */
   763→          }
   764→          resumeAfterAction();
   765→        })();
   766→        return true;
   767→      }
   768→
   769→      const target = store.models.find(
   770→        (m) => m.name.toLowerCase() === cmd.name.toLowerCase()
   771→      ) ?? store.models.find((m) => m.name.toLowerCase().includes(cmd.name.toLowerCase()));
   772→      if (!target) return false; // mentioned nothing we have → normal chat
   773→      store.removeModel(target.id);
   774→      void (async () => {
   775→        try {
   776→          await respond(`Removed the ${target.name}.`);
   777→        } catch {
   778→          /* best effort */
   779→        }
   780→        resumeAfterAction();
   781→      })();
   782→      return true;
   783→    },
   784→    [noteUser, pauseMic, respond, resumeAfterAction]
   785→  );
   786→
   787→  /* ---------------------------- one turn ---------------------------- */
   788→
   789→  const runTurn = useCallback(
   790→    async (userText: string) => {
   791→      if (stateRef.current === "thinking" || stateRef.current === "speaking") return;
   792→      if (tryWorkbench(userText)) return;
   793→      if (tryDelete(userText)) return;
   794→      if (tryBuild(userText)) return;
   795→      setError(null);
   796→      noteUser(userText);
   797→      setInterim("");
   798→      setAgentState("thinking");
   799→
   800→      const sessNow = sessionRef.current;
   801→      if (sessNow?.rec) {
   802→        try {
   803→          sessNow.rec.stop();
   804→        } catch {
   805→          /* noop */
   806→        }
   807→      }
   808→
   809→      let reply = "";
   810→      try {
   811→        reply = await askChat(userText);
   812→      } catch (err) {
   813→        if (!activeRef.current || abortRef.current?.signal.aborted) return;
   814→        const msg =
   815→          err instanceof Error && err.message ? err.message : "The AI service failed.";
   816→        setError(msg);
   817→        toast.error(msg, {
   818→          action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   819→        });
   820→        const s2 = sessionRef.current;
   821→        if (s2 && !s2.stopped) {
   822→          setAgentState("listening");
   823→          beginListeningRef.current(s2);
   824→        } else {
   825→          setAgentState("idle");
   826→        }
   827→        return;
   828→      }
   829→
   830→      if (!activeRef.current) return;
   831→      setLastReply(reply);
   832→
   833→      try {
   834→        await respond(reply);
   835→      } catch (err) {
   836→        if (!activeRef.current || abortRef.current?.signal.aborted) return;
   837→        const msg = err instanceof Error && err.message ? err.message : "Voice playback failed.";
   838→        setError(msg);
   839→        toast.error(msg);
   840→      }
   841→      if (!activeRef.current) return;
   842→
   843→      // Resume: voice sessions go back to listening; text sessions wait for input.
   844→      const sess = sessionRef.current;
   845→      if (sess && !sess.stopped) {
   846→        const delay = window.setTimeout(() => {
   847→          const s3 = sessionRef.current;
   848→          if (s3 === sess && !s3.stopped && activeRef.current) {
   849→            setAgentState("listening");
   850→            beginListeningRef.current(s3);
   851→          }
   852→        }, 350);
   853→        void delay;
   854→      } else {
   855→        setAgentState("idle");
   856→      }
   857→    },
   858→    [askChat, noteUser, respond, setAgentState, tryBuild, tryDelete, tryWorkbench]
   859→  );
   860→
   861→  /* --------------------- recognition wiring ------------------------- */
   862→
   863→  const setupRecognition = useCallback(
   864→    (mySess: VoiceSession) => {
   865→      const rec = new mySess.recCtor();
   866→      rec.continuous = true;
   867→      rec.interimResults = true;
   868→      rec.lang = "en-US";
   869→
   870→      const submitVoice = (text: string) => {
   871→        const t = text.trim();
   872→        if (!t) return;
   873→        if (mySess.interimTimer) {
   874→          clearTimeout(mySess.interimTimer);
   875→          mySess.interimTimer = null;
   876→        }
   877→        if (mySess.debounce) {
   878→          clearTimeout(mySess.debounce);
   879→          mySess.debounce = null;
   880→        }
   881→        mySess.finalBuf = "";
   882→        mySess.lastInterim = "";
   883→        setInterim("");
   884→        try {
   885→          mySess.rec?.stop();
   886→        } catch {
   887→          /* noop */
   888→        }
   889→        void runTurn(t);
   890→      };
   891→
   892→      rec.onresult = (event) => {
   893→        if (sessionRef.current !== mySess || mySess.stopped) return;
   894→        if (stateRef.current !== "listening") return;
   895→        let interimText = "";
   896→        for (let i = event.resultIndex; i < event.results.length; i++) {
   897→          const r = event.results[i];
   898→          if (r.isFinal) mySess.finalBuf += r[0].transcript + " ";
   899→          else interimText += r[0].transcript;
   900→        }
   901→        if (interimText) setInterim(interimText);
   902→        if (mySess.interimTimer) {
   903→          clearTimeout(mySess.interimTimer);
   904→          mySess.interimTimer = null;
   905→        }
   906→        mySess.lastInterim = interimText;
   907→
   908→        if (mySess.finalBuf.trim()) {
   909→          if (mySess.debounce) clearTimeout(mySess.debounce);
   910→          // Small pause after final words → send.
   911→          mySess.debounce = setTimeout(() => submitVoice(mySess.finalBuf), 850);
   912→        } else if (interimText.trim()) {
   913→          // Safety net: some browsers stall before emitting a final result.
   914→          mySess.interimTimer = setTimeout(
   915→            () => submitVoice(mySess.lastInterim),
   916→            1700
   917→          );
   918→        }
   919→      };
   920→
   921→      rec.onerror = (event) => {
   922→        if (sessionRef.current !== mySess || mySess.stopped) return;
   923→        const kind = event.error;
   924→        if (kind === "no-speech" || kind === "aborted") return;
   925→        if (kind === "not-allowed" || kind === "service-not-allowed") {
   926→          fallbackToText(
   927→            "Microphone access was blocked — open this app in its own browser tab to talk out loud, or type below."
   928→          );
   929→          return;
   930→        }
   931→        if (kind === "network") {
   932→          fallbackToText(
   933→            "Speech recognition lost its connection — keep talking by typing below."
   934→          );
   935→        }
   936→      };
   937→
   938→      rec.onend = () => {
   939→        if (sessionRef.current !== mySess || mySess.stopped) return;
   940→        if (stateRef.current === "listening" && modeRef.current === "voice") {
   941→          beginListeningRef.current(mySess);
   942→        }
   943→      };
   944→
   945→      mySess.rec = rec;
   946→    },
   947→    [fallbackToText, runTurn]
   948→  );
   949→
   950→  /* ------------------------------ start ----------------------------- */
   951→
   952→  const start = useCallback(() => {
   953→    if (activeRef.current) return;
   954→    const s = useInfinity.getState().settings;
   955→    if (!isConfigured(s)) {
   956→      toast("Add your API key in Settings to start talking to Infinity.", {
   957→        action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
   958→      });
   959→      onNeedSettingsRef.current();
   960→      return;
   961→    }
   962→    const recCtor = getSpeechRecognitionCtor();
   963→    if (!recCtor) {
   964→      toast("Voice input isn't supported in this browser — type instead (press /).");
   965→      fallbackToText(
   966→        "Voice input isn't supported in this browser — type below and Infinity will answer out loud."
   967→      );
   968→      return;
   969→    }
   970→
   971→    const sess: VoiceSession = {
   972→      stopped: false,
   973→      micStream: null,
   974→      micAnalyser: null,
   975→      micData: null,
   976→      rec: null,
   977→      recCtor,
   978→      finalBuf: "",
   979→      lastInterim: "",
   980→      interimTimer: null,
   981→      debounce: null,
   982→      lastRestart: 0,
   983→    };
   984→    sessionRef.current = sess;
   985→    activeRef.current = true;
   986→    modeRef.current = "voice";
   987→    setMode("voice");
   988→    setSessionActive(true);
   989→    setMicBlocked(false);
   990→    setError(null);
   991→    setLastUser("");
   992→    setLastReply("");
   993→    historyRef.current = [];
   994→
   995→    void (async () => {
   996→      try {
   997→        let ctx = ctxRef.current;
   998→        if (!ctx) {
   999→          ctx = new AudioContext();
  1000→          ctxRef.current = ctx;
  1001→        }
  1002→        try {
  1003→          await ctx.resume();
  1004→        } catch {
  1005→          /* noop */
  1006→        }
  1007→        sess.micStream = await navigator.mediaDevices.getUserMedia({
  1008→          audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
  1009→        });
  1010→        if (sessionRef.current !== sess) {
  1011→          sess.micStream.getTracks().forEach((t) => t.stop());
  1012→          return;
  1013→        }
  1014→        const micSrc = ctx.createMediaStreamSource(sess.micStream);
  1015→        const an = ctx.createAnalyser();
  1016→        an.fftSize = 512;
  1017→        micSrc.connect(an); // deliberately not connected to destination
  1018→        sess.micAnalyser = an;
  1019→        sess.micData = new Uint8Array(new ArrayBuffer(an.fftSize));
  1020→
  1021→        setupRecognition(sess);
  1022→        setAgentState("listening");
  1023→        beginListening(sess);
  1024→      } catch {
  1025→        if (sessionRef.current !== sess) return;
  1026→        fallbackToText(
  1027→          "Microphone unavailable — open this app in its own browser tab and allow the mic, or type below."
  1028→        );
  1029→        toast.error("Microphone unavailable — you can type instead.");
  1030→      }
  1031→    })();
  1032→  }, [beginListening, fallbackToText, setAgentState, setupRecognition]);
  1033→
  1034→  /* ----------------------------- typing ----------------------------- */
  1035→
  1036→  const sendText = useCallback(
  1037→    (text: string) => {
  1038→      const t = text.trim();
  1039→      if (!t) return;
  1040→      // TYPING MODE: sending text always implies a silent text session —
  1041→      // no mic, no sound, replies appear as text. If a VOICE session is
  1042→      // already running, the typed text joins it instead (barge-in, spoken).
  1043→      if (!activeRef.current) {
  1044→        activeRef.current = true;
  1045→        modeRef.current = "text";
  1046→        setMode("text");
  1047→        setSessionActive(true);
  1048→        setMicBlocked(false);
  1049→        setError(null);
  1050→        setLastUser("");
  1051→        setLastReply("");
  1052→        setTranscript([]);
  1053→        historyRef.current = [];
  1054→        setAgentState("idle");
  1055→      }
  1056→      // Workbench commands are local — they work even without an API key.
  1057→      if (tryWorkbench(t)) return;
  1058→      if (tryDelete(t)) return;
  1059→      if (tryBuild(t)) return;
  1060→      const s = useInfinity.getState().settings;
  1061→      if (!isConfigured(s)) {
  1062→        toast("Add your API key in Settings first.", {
  1063→          action: { label: "Settings", onClick: () => onNeedSettingsRef.current() },
  1064→        });
  1065→        onNeedSettingsRef.current();
  1066→        return;
  1067→      }
  1068→      void runTurn(t);
  1069→    },
  1070→    [runTurn, setAgentState, tryBuild, tryDelete, tryWorkbench]
  1071→  );
  1072→
  1073→  const toggle = useCallback(() => {
  1074→    if (activeRef.current) stop();
  1075→    else start();
  1076→  }, [start, stop]);
  1077→
  1078→  /* --------------------------- level loop --------------------------- */
  1079→
  1080→  useEffect(() => {
  1081→    let raf = 0;
  1082→    const frame = (t: number) => {
  1083→      const st = stateRef.current;
  1084→      let target = 0;
  1085→      const sess = sessionRef.current;
  1086→      if (st === "listening" && sess?.micAnalyser && sess.micData) {
  1087→        target = rms(sess.micAnalyser, sess.micData);
  1088→      } else if (st === "speaking") {
  1089→        if (playbackAnalyserRef.current && playbackDataRef.current) {
  1090→          target = rms(playbackAnalyserRef.current, playbackDataRef.current);
  1091→        } else if (syntheticUntilRef.current > Date.now()) {
  1092→          target = 0.32 + 0.28 * Math.sin(t * 0.012);
  1093→        }
  1094→      }
  1095→      const prev = levelRef.current;
  1096→      levelRef.current = prev + (target - prev) * (target > prev ? 0.55 : 0.12);
  1097→      raf = requestAnimationFrame(frame);
  1098→    };
  1099→    raf = requestAnimationFrame(frame);
  1100→    return () => cancelAnimationFrame(raf);
  1101→  }, []);
  1102→
  1103→  // stop everything on unmount
  1104→  useEffect(() => () => stopRef.current(), []);
  1105→
  1106→  return {
  1107→    state,
  1108→    mode,
  1109→    sessionActive,
  1110→    interim,
  1111→    lastUser,
  1112→    lastReply,
  1113→    error,
  1114→    micBlocked,
  1115→    transcript,
  1116→    building,
  1117→    levelRef,
  1118→    start,
  1119→    stop,
  1120→    toggle,
  1121→    sendText,
  1122→  };
  1123→}
